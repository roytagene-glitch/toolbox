local B64CHARS = "ABCDEFGHIJKLMNOPQRSTUVWXYZabcdefghijklmnopqrstuvwxyz0123456789+/"
local function b64decode(data)
	local lookup = {}
	for i = 1, #B64CHARS do lookup[B64CHARS:byte(i)] = i - 1 end
	local out, n, buf, bits = {}, 0, 0, 0
	for i = 1, #data do
		local v = lookup[data:byte(i)]
		if v then
			buf = buf * 64 + v; bits += 6
			if bits >= 8 then
				bits -= 8
				local byte = math.floor(buf / 2 ^ bits)
				buf = buf % 2 ^ bits
				n += 1; out[n] = string.char(byte)
			end
		end
	end
	return table.concat(out)
end

local function draggable(handle, target, onTap)
	local dragging, start, startPos, moved = false, nil, nil, 0
	handle.InputBegan:Connect(function(i)
		if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
			dragging = true; moved = 0; start = i.Position; startPos = target.Position
			i.Changed:Connect(function()
				if i.UserInputState == Enum.UserInputState.End then
					dragging = false
					if onTap and moved < 8 then onTap() end
				end
			end)
		end
	end)
	UserInputService.InputChanged:Connect(function(i)
		if dragging and (i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch) then
			local d = i.Position - start
			moved = math.max(moved, math.abs(d.X) + math.abs(d.Y))
			target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
end

local function httpGet(url)
	local fn = (syn and syn.request) or (http and http.request) or http_request or request
	if fn then
		local ok, r = pcall(fn, { Url = url, Method = "GET" })
		if ok and r and r.Body then return r.Body end
	end
	local ok, b = pcall(function() return game:HttpGet(url) end)
	return ok and b or nil
end
local function decodeJson(s)
	local ok, t = pcall(function() return HttpService:JSONDecode(s) end)

end
local function normalize(raw)
	if not raw then return nil end
	local id, name
	if raw.Asset then id = raw.Asset.Id or raw.Asset.id; name = raw.Asset.Name or raw.Asset.name
	elseif raw.asset then id = raw.asset.id; name = raw.asset.name
	else id = raw.id or raw.Id; name = raw.name or raw.Name end
	if not id then return nil end
	return { id = tonumber(id) or id, name = name or ("Asset " .. tostring(id)) }
end
local function searchAssets(cat, kw, page)
	local res = {}
	local url = ("https://apis.roblox.com/toolbox-service/v1/marketplace/%d?category=%d&keyword=%s&limit=30&sortType=0&page=%d")
		:format(cat.id, cat.id, HttpService:UrlEncode(kw or ""), page)
	local body = httpGet(url)
	local d = body and decodeJson(body)
	if d and d.data then
		for _, r in ipairs(d.data) do
			local n = normalize(r)
			if n then table.insert(res, n) end
		end
	end
	if #res == 0 and page == 1 then
		local url2 = ("https://catalog.roblox.com/v1/search/items/details?Category=%s&Keyword=%s&Limit=30&sortType=0")
			:format(cat.key, HttpService:UrlEncode(kw or ""))
		local b2 = httpGet(url2)
		local d2 = b2 and decodeJson(b2)
		if d2 and d2.data then
			for _, r in ipairs(d2.data) do
				local n = normalize(r)
				if n then table.insert(res, n) end
			end
		end
	end
	return res
end

local baseEnv = getfenv()
local loadStr = loadstring or (syn and syn.load) or (getgenv and getgenv().loadstring)
local realRequire = require
local BAD = { "failed to decompile", "not found", "decompiler error", "decompilation failed", "unable to decompile" }
local function goodSrc(src)
	if type(src) ~= "string" or #src < 3 then return false end
	if #src < 160 then
		local low = src:lower()
		for _, b in ipairs(BAD) do if low:find(b, 1, true) then return false end end
	end
	return true
end
local function readSource(sc)
	for attempt = 1, 2 do
		local src
		pcall(function() src = sc.Source end)
		if goodSrc(src) then return src end
		if decompile then local ok, r = pcall(decompile, sc); if ok and goodSrc(r) then return r end end
		if getscriptsource then local ok, r = pcall(getscriptsource, sc); if ok and goodSrc(r) then return r end end
		if gethiddenproperty then local ok, r = pcall(gethiddenproperty, sc, "Source"); if ok and goodSrc(r) then return r end end
		if attempt < 2 then task.wait(0.15) end
	end
end

local customRequire
local function makeEnv(sc)
	return setmetatable({ script = sc, owner = player, require = function(m) return customRequire(m) end }, { __index = baseEnv })
end
local function compileScript(sc)
	local fn
	local src = readSource(sc)
	if src and loadStr then
		local f = loadStr(src, "=" .. sc.Name)
		if f and pcall(setfenv, f, makeEnv(sc)) then fn = f end
	end
	if not fn and getscriptclosure then
		pcall(function()
			local f = getscriptclosure(sc)
			if f then pcall(setfenv, f, makeEnv(sc)); fn = f end
		end)
	end
	return fn
end
local moduleCache = {}
customRequire = function(m)
	if typeof(m) == "Instance" and m:IsA("ModuleScript") then
		if moduleCache[m] == nil then
			local fn = compileScript(m)
			if not fn then
				local ok, res = pcall(realRequire, m)
				if ok then return res end
				error("Dhafin: gagal membaca ModuleScript " .. m.Name, 2)
			end
			local ok, res = pcall(fn)
			if not ok then error(res, 2) end
			moduleCache[m] = { res }
		end
		return moduleCache[m][1]
	end
	return realRequire(m)
end

local function runScripts(list)
	local ran, failed = 0, 0
	for _, sc in ipairs(list) do
		local fn = compileScript(sc)
		if fn then
			pcall(function() sc.Disabled = false end)
			task.spawn(function()
				local ok, err = pcall(fn)
				if not ok then setStatus("[SCRIPT ERR] " .. sc.Name .. ": " .. tostring(err), C.RED) end
			end)
			ran += 1
		else
			failed += 1
		end
	end
	return ran, failed
end

local function loadModel(id)
	local best, bestN, lastErr = nil, -1, nil
	local ok1, res = pcall(function() return game:GetObjects("rbxassetid://" .. tostring(id)) end)
	if ok1 and res and #res > 0 then
		local c = Instance.new("Model"); c.Name = "DhafinLoaded"
		for _, ch in ipairs(res) do ch.Parent = c end
		best, bestN = c, countScripts(c)
	else
		lastErr = res
	end
	local m
	local ok2, err = pcall(function() m = InsertService:LoadAsset(tonumber(id) or id) end)
	if ok2 and m then
		local n = countScripts(m)
		if n > bestN then
			if best then best:Destroy() end
			best, bestN = m, n
		else
			m:Destroy()
		end
	else
		lastErr = lastErr or err
	end
	return best, lastErr
end

local activeSound
local function stopAudio()
	if activeSound then pcall(function() activeSound:Destroy() end); activeSound = nil end
end

local function camFront(dist)
	local cam = Workspace.CurrentCamera
	return cam and (cam.CFrame * CFrame.new(0, 0, -dist)) or CFrame.new(0, 10, 0)
end

local function insertAsset(id, cat, name)
	name = name or ("Asset " .. tostring(id))
	setStatus("Memasukkan: " .. name, C.SUB)
	local key = cat.key

	if key == "Audio" then
		stopAudio()
		local s = mk("Sound", { Name = name, SoundId = "rbxassetid://" .. id, Volume = 1 }, SoundService)
		activeSound = s
		local t0 = tick()
		while not s.IsLoaded and tick() - t0 < 6 do task.wait(0.1) end
		pcall(function() s:Play() end)
		setStatus("♪ Memutar: " .. name, C.GREEN)
		return true
	end

	if key == "Decal" or key == "Texture" then
		local p = mk("Part", { Name = name, Size = Vector3.new(4, 4, 0.5), Anchored = true, CFrame = camFront(10) }, Workspace)
		for _, face in ipairs({ "Front", "Back", "Top", "Bottom", "Left", "Right" }) do
			mk("Decal", { Texture = "rbxassetid://" .. id, Face = Enum.NormalId[face] }, p)
		end
		setStatus("Decal dipasang: " .. name, C.GREEN)
		return true
	end

	if key == "Mesh" then
		local p = mk("Part", { Name = name, Size = Vector3.new(4, 4, 4), Anchored = true, CFrame = camFront(12) }, Workspace)
		mk("SpecialMesh", { MeshType = Enum.MeshType.FileMesh, MeshId = "rbxassetid://" .. id }, p)
		setStatus("Mesh dipasang: " .. name, C.GREEN)
		return true
	end

	if key == "Animation" then
		local hum = player.Character and player.Character:FindFirstChildOfClass("Humanoid")
		if not hum then setStatus("[!] Karakter tidak ditemukan", C.RED); return false end
		local anim = mk("Animation", { Name = name, AnimationId = "rbxassetid://" .. id }, nil)
		local ok = pcall(function() hum:LoadAnimation(anim):Play() end)
		setStatus(ok and ("Animasi diputar: " .. name) or "[!] Animasi gagal diputar", ok and C.GREEN or C.RED)
		return ok
	end

	local model, err = loadModel(id)
	if not model then setStatus("[ERROR] " .. tostring(err), C.RED); return false end

	local scripts, parts = {}, 0
	for _, d in ipairs(model:GetDescendants()) do
		pcall(function()
			if d:IsA("BasePart") then
				parts += 1
				if S.opts.anchor then d.Anchored = true end
				if S.opts.hifi and (d:IsA("MeshPart") or d:IsA("UnionOperation")) then
					d.RenderFidelity = Enum.RenderFidelity.Precise
				end
			elseif d:IsA("LocalScript") or d:IsA("Script") then
				table.insert(scripts, d)
			end
		end)
	end

	model.Name = name
	model.Parent = Workspace
	pcall(function() model:PivotTo(camFront(15)) end)
	local total = countScripts(model)
	setStatus(("Disisipkan: %s • %d part • %d script"):format(name, parts, total), C.GREEN)

	if #scripts > 0 and S.opts.autorun then
		if not loadStr then
			setStatus("[!] Executor tanpa loadstring - script tidak dijalankan", C.RED)
		else
			local ran, failed = runScripts(scripts)
			if failed > 0 then
				setStatus(("%d script jalan, %d tidak terbaca (script server tidak dikirim Roblox ke client)"):format(ran, failed), C.RED)
			else
				setStatus(("Disisipkan: %s • %d script dijalankan"):format(name, ran), C.GREEN)
			end
		end
	end
	return true
end

local function clearGrid()
	for _, c in ipairs(grid:GetChildren()) do
		if c:IsA("GuiObject") then c:Destroy() end
	end
	S.cards = {}
end

local function notice(text, order)
	local f = mk("Frame", { BackgroundTransparency = 1, LayoutOrder = order or 0, Size = UDim2.new(1, 0, 0, 108) }, grid)
	local l = txt(f, text, 11, C.SUB, Enum.Font.Gotham, Enum.TextXAlignment.Center)
	l.Size = UDim2.fromScale(1, 1); l.TextWrapped = true
	return f
end

local function addTile(item, order)
	local t = mk("Frame", { BackgroundColor3 = C.TILE, BorderSizePixel = 0, LayoutOrder = order }, grid)
	round(t, 8); line(t)
	local img = mk("ImageLabel", {
		Size = UDim2.new(1, -12, 0, 56), Position = UDim2.fromOffset(6, 6), BackgroundColor3 = C.PANEL,
		BorderSizePixel = 0, Image = "rbxthumb://type=Asset&id=" .. tostring(item.id) .. "&w=150&h=150",
		ScaleType = Enum.ScaleType.Fit,
	}, t)
	round(img, 6)
	local nm = txt(t, item.name, 10, C.TEXT, Enum.Font.GothamBold)
	nm.Position = UDim2.fromOffset(6, 65); nm.Size = UDim2.new(1, -12, 0, 14); nm.TextTruncate = Enum.TextTruncate.AtEnd

	local b = button(t, "Insert", 11, C.BLUE)
	b.Size = UDim2.new(1, -12, 0, 20); b.Position = UDim2.new(0, 6, 1, -26)
	round(b, 5); hov(b, C.BLUE, C.BLUE2)
	local busy = false
	b.MouseButton1Click:Connect(function()
		if busy then return end
		busy = true; b.Text = "..."
		task.spawn(function()
			local ok = insertAsset(item.id, S.cat, item.name)
			b.Text = ok and "✓" or "✗"
			b.BackgroundColor3 = ok and C.GREEN or C.RED
			task.wait(1.2)
			b.Text = "Insert"; b.BackgroundColor3 = C.BLUE; busy = false
		end)
	end)
	return { item = item, nameLbl = nm }
end

local function enrichNames(cards, token)
	task.spawn(function()
		for _, c in ipairs(cards) do
			if token ~= S.token or not c.nameLbl.Parent then return end
			local nm = getAssetName(c.item.id)
			if nm then c.item.name = nm; c.nameLbl.Text = nm end
			task.wait(0.03)
		end
	end)
end

local runSearch
runSearch = function(append)
	S.token += 1
	local my = S.token
	if not append then
		S.page = 1
		clearGrid()
	else
		for _, c in ipairs(grid:GetChildren()) do
			if c.Name == "MoreTile" then c:Destroy() end
		end
	end
	local ph = notice("⏳ Mengambil data...", 100000)
	setStatus("Mencari " .. S.cat.label .. (S.kw ~= "" and (" • " .. S.kw) or "") .. "...", C.SUB)
	task.spawn(function()
		local list = searchAssets(S.cat, S.kw, S.page)
		if my ~= S.token then return end
		pcall(function() ph:Destroy() end)
		if #list == 0 and not append then
			notice("Tidak ada hasil.\nCoba kata lain atau pakai Asset ID di bawah.", 1)
			setStatus("Tidak ada hasil", C.RED)
			return
		end
		local newCards = {}
		local base = #S.cards
		for i, it in ipairs(list) do
			local card = addTile(it, base + i)
			table.insert(S.cards, card); table.insert(newCards, card)
		end
		if #list >= 20 then
			local more = button(grid, "Muat lagi", 11, C.PANEL, C.BLUE2)
			more.Name = "MoreTile"; more.LayoutOrder = 99999
			round(more, 8); line(more, C.BLUE); hov(more, C.PANEL, C.TILE)
			more.MouseButton1Click:Connect(function()
				S.page += 1
				runSearch(true)
			end)
		end
		setStatus(("Ditemukan %d hasil"):format(#S.cards), C.GREEN)
		enrichNames(newCards, my)
	end)
end
